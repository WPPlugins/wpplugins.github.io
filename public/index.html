    5. Use with Composer (composer.json)

    Because most WordPress.org plugins do not ship a composer.json, declare a package repository pointing at WPPlugins/<slug> together with composer/installers so Composer installs plugins directly into wp-content/plugins/.

    composer.json 
    {
  "repositories": [
    {
      "type": "package",
      "package": {
        "name": "wpplugins/<slug>",
        "version": "1.2.3",
        "type": "wordpress-plugin",
        "source": {
          "url": "https://github.com/WPPlugins/<slug>.git",
          "type": "git",
          "reference": "1.2.3"
        },
        "require": {
          "composer/installers": "^2.0"
        }
      }
    }
  ],
  "require": {
    "wpplugins/<slug>": "1.2.3"
  },
  "extra": {
    "installer-paths": {
      "wp-content/plugins/{$name}/": ["type:wordpress-plugin"]
    }
  }
}
    Install with Composer 
    composer install
